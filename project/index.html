terraclean_firmware.ino 
#include <LiquidCrystal_I2C.h>

const int TRIG_PIN = 9;
const int ECHO_PIN = 10;
const int BUZZER_PIN = 8;
const int LED_PIN = 13;

const int BIN_HEIGHT_CM = 50;
const int CRITICAL_DIST_CM = 10;
const int WARNING_DIST_CM = 20;

LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
  Serial.begin(9600);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  lcd.init();
  lcd.backlight();
}

void loop() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH);
  int distanceCm = duration * 0.034 / 2;

  int fillPercent = map(BIN_HEIGHT_CM - distanceCm, 0, BIN_HEIGHT_CM, 0, 100);
  fillPercent = constrain(fillPercent, 0, 100);

  lcd.setCursor(0, 0);
  lcd.print("Fill: "); lcd.print(fillPercent); lcd.print("%   ");

  if (distanceCm <= CRITICAL_DIST_CM) {
    lcd.setCursor(0, 1); lcd.print("Status: CRITICAL!");
    digitalWrite(BUZZER_PIN, HIGH);
    digitalWrite(LED_PIN, HIGH);
  } else if (distanceCm <= WARNING_DIST_CM) {
    lcd.setCursor(0, 1); lcd.print("Status: WARNING ");
    digitalWrite(BUZZER_PIN, LOW);
    digitalWrite(LED_PIN, LOW);
  } else {
    lcd.setCursor(0, 1); lcd.print("Status: NORMAL  ");
    digitalWrite(BUZZER_PIN, LOW);
    digitalWrite(LED_PIN, LOW);
  }
  delay(500);
}

What Makes TerraClean AI Different?
Key innovation pillars uniting smart IoT and traditional material science

Real-time waste overflow detection

Continuous distance measurement using HC-SR04 ultrasonic sensor alerts administrators before waste spills out.

Natural clay/terracotta-based odor-control concept

Uses porous terracotta clay material properties to help absorb ambient moisture and mitigate foul smells naturally.

Charcoal-based odor adsorption layer

Activated charcoal acts as a high-surface-area porous filter to adsorb volatile compounds that cause unpleasant odors.

Combines smart sensing with traditional material knowledge

Bridges modern low-cost electronics with sustainable eco-friendly material wisdom for practical smart city use.

Designed to improve user acceptance of public waste bins

Minimizing foul odors around public bins encourages citizens to deposit waste properly inside the bin rather than outside.

Modular and replaceable bio-filter concept

Designed as a replaceable cassette tray at the bottom of the bin for easy maintenance by municipal staff.
